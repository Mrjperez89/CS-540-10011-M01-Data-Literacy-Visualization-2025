# To Do: Convert numeric columns to proper formats (removing commas and extra spaces)

# Some datasets store numbers as text (ex: "12,345" or "  98,765 ").
# This loop looks for columns that are likely numeric and converts them safely.

# Identify object (string) columns that look like they contain numbers.
candidate_cols = []
for col in filtered_df.columns:
    if filtered_df[col].dtype == "object":
        # Grab a small sample of non-null values to test conversion.
        sample = filtered_df[col].dropna().astype(str).head(25)

        # If most sampled values can be converted after cleaning, treat it as numeric.
        # NOTE: This dataset includes currency-like values (ex: "$72,095,003.00"), so I strip "$" too.
        cleaned = (
            sample.str.replace(",", "", regex=False)
                  .str.replace("$", "", regex=False)
                  .str.strip()
        )
        convertible_ratio = pd.to_numeric(cleaned, errors="coerce").notna().mean() if len(cleaned) > 0 else 0

        if convertible_ratio >= 0.8:  # 80%+ convertible is a good signal
            candidate_cols.append(col)

# Convert the detected numeric-like text columns
converted_cols = []
for col in candidate_cols:
    filtered_df[col] = (
        filtered_df[col]
        .astype(str)
        .str.replace(",", "", regex=False)
        .str.replace("$", "", regex=False)
        .str.strip()
    )
    filtered_df[col] = pd.to_numeric(filtered_df[col], errors="coerce")
    converted_cols.append(col)

print("Converted numeric-like columns:")
print(converted_cols if converted_cols else "None detected")
